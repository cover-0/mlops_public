```python
"""
문제 1 (1주 + 2주). 공공 API 대기질 수집 - 수정본
=========================================
수정 사항:
1. 페이징 결함 해결: 첫 페이지만 가져오던 것을 전체 건수(totalCount)에 도달할 때까지 반복 수집
2. 결측치 처리 결함 해결: 결측치('-')를 임의로 0으로 대체하지 않고 유효한 측정값만으로 평균 산출
3. 최고 측정소 선정: 전체 유효 측정소 중에서 최고 지수 측정소 탐색 (동점 시 이름 오름차순)
"""

import json

from common import PRACTICE, key, parse_id, pick_int, pick_subset

sid = parse_id()
SRC = PRACTICE / "chapter2" / "data" / "input" / "airquality_seoul_2200.json"
ITEMS = json.loads(SRC.read_text(encoding="utf-8"))["response"]["body"]["items"]

PAGE_SIZE = pick_int(sid, "q1-page", 12, 28)
for i in pick_subset(sid, "q1-miss", len(ITEMS), pick_int(sid, "q1-nmiss", 2, 6)):
    ITEMS[i]["khaiValue"] = "-"


def fetch_page(page_no: int) -> dict:
    """API 한 페이지를 돌려준다. 실제 API와 같은 header/body 두 계층이다."""
    start = (page_no - 1) * PAGE_SIZE
    return {
        "header": {"resultCode": "00", "resultMsg": "NORMAL_CODE"},
        "body": {
            "totalCount": len(ITEMS), "pageNo": page_no, "numOfRows": PAGE_SIZE,
            "items": ITEMS[start:start + PAGE_SIZE],
        },
    }


# [결함 1 수정] 전체 건수(totalCount)까지 모든 페이지 순회 수집
all_rows = []
page_no = 1
total_count = None

while True:
    resp = fetch_page(page_no)
    header, body = resp["header"], resp["body"]
    if header["resultCode"] != "00":
        raise SystemExit(f"API 오류: {header['resultMsg']}")

    if total_count is None:
        total_count = body["totalCount"]

    items = body.get("items", [])
    if not items:
        break
    all_rows.extend(items)

    if len(all_rows) >= total_count:
        break
    page_no += 1

n_collected = len(all_rows)
complete = 1 if n_collected == total_count else 0

# [결함 2 수정] 결측치('-')를 0으로 왜곡하지 않고 분리 집계
missing_rows = [r for r in all_rows if r["khaiValue"] == "-"]
valid_rows = [r for r in all_rows if r["khaiValue"] != "-"]

n_missing_khai = len(missing_rows)
valid_values = [int(r["khaiValue"]) for r in valid_rows]
mean_khai = sum(valid_values) / len(valid_values) if valid_values else 0.0

# [결함 3 수정] 전체 유효 측정소 기준 최고 측정소 탐색
top = sorted(valid_rows, key=lambda r: (-int(r["khaiValue"]), r["stationName"]))[0]
max_khai_station = top["stationName"]

print(f"응답 코드 {header['resultCode']} ({header['resultMsg']})")
print(f"수집 {n_collected}/{total_count}건 -- {'수집 완료' if complete else '수집 미완료'}")
print(f"통합대기환경지수 평균 {mean_khai:.2f}, 최고 {max_khai_station} ({top['khaiValue']})")

key("total_count", total_count)
key("n_collected", n_collected)
key("complete", complete)
key("n_missing_khai", n_missing_khai)
key("mean_khai", float(mean_khai))
key("max_khai_station", max_khai_station)
```

### 수정한 코드 전문 (`q2_fixed.py`)

```python
"""
문제 2 (3주). 같은 집계를 두 계층에 두었다 - 수정본
===========================================
수정 사항:
1. 결함 원인 분석: 배치 계층은 지연 허용(GRACE_BATCH)이 15분인 반면, 
   스피드 계층은 GRACE_SPEED(8분)로 서로 다르게 설정되어 있어,
   72분 지연된 이벤트 20건(각 창당 10건)이 스피드 계층에서만 탈락하는 Lambda skew 발생.
2. 결함 수정: 스피드 계층의 집계 규칙(지연 허용 시간)을 배치 계층과 동일하게 15분으로 통일(또는 단일 규칙 사용).
3. 수정 전 스피드 계층 결과와 비교하여 n_diff_before, missing_events_before, max_mean_gap_before 산출 및
   수정 후 sha_batch, sha_speed, sha_equal 출력.
"""

import hashlib
import json
from datetime import datetime, timedelta

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter3" / "data" / "input"
FILES = ["airquality_seoul_2200.json", "airquality_seoul_current.json"]
WINDOW_MIN = 60
GRACE_BATCH = 15                                   # 배치 계층의 지연 허용(분)
GRACE_SPEED = pick_choice(sid, "q2-grace", [5, 8, 10, 12])   # 수정 전 스피드 계층의 지연 허용(분)
LATE_PERIOD = pick_int(sid, "q2-period", 4, 7)     # 정렬 순번이 이 값의 배수인 이벤트가 72분 늦게 도착


def load_events():
    rows = []
    for name in FILES:
        body = json.loads((INPUT_DIR / name).read_text(encoding="utf-8"))["response"]["body"]
        for item in body["items"]:
            if item.get("pm10Value") in (None, "-", ""):
                continue
            rows.append({
                "station": item["stationName"],
                "event_time": datetime.strptime(item["dataTime"], "%Y-%m-%d %H:%M"),
                "pm10": int(item["pm10Value"]),
            })
    rows.sort(key=lambda r: (r["event_time"], r["station"]))
    for i, row in enumerate(rows):
        delay = 72 if (i + 1) % LATE_PERIOD == 0 else 3
        row["arrival_time"] = row["event_time"] + timedelta(minutes=delay)
    return rows


def window_of(t):
    return t.replace(minute=0, second=0, microsecond=0)


def aggregate(rows, grace_min):
    """지연 허용 시간(grace_min)을 인자로 받아 창별 건수와 PM10 평균을 집계하는 단일 공통 함수"""
    buckets = {}
    for row in rows:
        win = window_of(row["event_time"])
        if row["arrival_time"] >= win + timedelta(minutes=WINDOW_MIN + grace_min):
            continue
        buckets.setdefault(win, []).append(row["pm10"])
    return {w.strftime("%Y-%m-%d %H:%M"): {"count": len(v), "pm10_mean": round(sum(v) / len(v), 2)}
            for w, v in sorted(buckets.items())}


def sha(obj):
    return hashlib.sha256(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode("utf-8")).hexdigest()[:16]


events = load_events()

# 배치 계층 집계 (15분 허용)
batch = aggregate(events, GRACE_BATCH)

# 수정 전 스피드 계층 집계 (학번별 GRACE_SPEED)
speed_before = aggregate(events, GRACE_SPEED)

# 수정 전 두 계층 차이 계산
diff_windows = []
missing_events_before = 0
mean_gaps = []

for win in sorted(set(batch) | set(speed_before)):
    b = batch.get(win, {"count": 0, "pm10_mean": 0.0})
    s = speed_before.get(win, {"count": 0, "pm10_mean": 0.0})
    if b != s:
        diff_windows.append(win)
    missing_events_before += (b["count"] - s["count"])
    mean_gaps.append(abs(round(b["pm10_mean"] - s["pm10_mean"], 2)))

n_diff_before = len(diff_windows)
max_mean_gap_before = max(mean_gaps) if mean_gaps else 0.0

# [결함 수정] 스피드 계층의 지연 허용을 배치와 동일하게 15분으로 일치
speed_after = aggregate(events, GRACE_BATCH)

sha_b = sha(batch)
sha_s = sha(speed_after)
sha_equal = 1 if sha_b == sha_s else 0

print(f"이벤트 {len(events)}건, 창 길이 {WINDOW_MIN}분")
print(f"[수정 전] 어긋난 창 {n_diff_before}개, 누락 이벤트 {missing_events_before}건, 최대 평균 차이 {max_mean_gap_before}")
print(f"[수정 후] 배치 sha: {sha_b}, 스피드 sha: {sha_s}, 일치 여부: {sha_equal}")
print(f"{'창':<18}{'배치 건수':>10}{'배치 평균':>11}{'스피드 건수':>12}{'스피드 평균':>12}")
for win in sorted(set(batch) | set(speed_after)):
    b = batch.get(win, {"count": 0, "pm10_mean": 0.0})
    s = speed_after.get(win, {"count": 0, "pm10_mean": 0.0})
    print(f"{win:<18}{b['count']:>10}{b['pm10_mean']:>11.2f}{s['count']:>12}{s['pm10_mean']:>12.2f}")

key("n_diff_before", n_diff_before)
key("missing_events_before", missing_events_before)
key("max_mean_gap_before", max_mean_gap_before)
key("sha_batch", sha_b)
key("sha_speed", sha_s)
key("sha_equal", sha_equal)
```

### 답안

>> 답: 원본 코드 q2.py의 26번째 줄(GRACE_SPEED)과 70번째 줄의 지연 필터링 조건문에 결함이 있다. 운영팀 보고와 달리 배치 계층의 지연 허용은 15분인데 스피드 계층은 8분으로 다르게 설정되었다.이로 인해 72분 지연된 이벤트 20건이 배치 계층(마감 75분)에서는 정상 집계되었으나, 스피드 계층(마감 68분)에서는 마감 시간을 초과해 모두 폐기되면서 두 계층 결과가 어긋나는 침묵 실패가 발생했다.

```python
"""
문제 3 (4주). 소비자 장애 뒤에 적재된 행 수 - 수정본
============================================
수정 사항:
1. 결함 원인 분석:
   - 결함 1 (L71): "적재 행 수 >= 원천 건수"를 근거로 exactly-once라고 잘못 판정. 실제로는 A 규칙(처리 후 커밋)에서
     커밋되지 않은 2건이 재시작 후 중복 적재되어 32행이 적재된 at-least-once 상태임.
   - 결함 2: At-least-once 파이프라인에서 중복을 걸러내는 사후 중복 제거(deduplication) 로직이 누락됨.
2. 결함 수정:
   - 고유 식별자(event_id) 기반의 중복 제거를 수행하여 중복 건수와 정제 후 행 수 산출.
   - B 규칙(읽은 직후 커밋)을 함께 시뮬레이션하여 유실된 고유 이벤트 수(lost_rule_b) 집계.
"""

import json

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
STREAM = PRACTICE / "chapter4" / "data" / "input" / "ch4_alo_duplicate_stream.jsonl"

ids = []
for line in STREAM.read_text(encoding="utf-8").splitlines():
    event_id = json.loads(line)["event_id"]
    if event_id not in ids:
        ids.append(event_id)
N = len(ids)                                              # 원천 이벤트 수 (30)

COMMIT_EVERY = pick_choice(sid, "q3-commit", [3, 4, 5, 6, 8])
_base = pick_int(sid, "q3-crash", 14, 24) // COMMIT_EVERY * COMMIT_EVERY
CRASH_AT = _base + 2 + pick_int(sid, "q3-rest", 0, COMMIT_EVERY - 3)   # 이 개수만큼 처리한 직후 장애
if CRASH_AT > N - 2:
    CRASH_AT -= COMMIT_EVERY


def run_rule_a():
    """A. 처리한 뒤 커밋 (At-Least-Once). 장애 뒤에는 마지막 커밋 위치부터 다시 읽는다."""
    rows, committed = [], 0
    for off in range(CRASH_AT):                           # 장애 전까지
        rows.append((off, ids[off]))
        if (off + 1) % COMMIT_EVERY == 0:
            committed = off + 1
    for off in range(committed, N):                       # 재시작 뒤
        rows.append((off, ids[off]))
    return rows


def run_rule_b():
    """B. 읽은 직후 커밋 (At-Most-Once). 장애 뒤에는 커밋된 위치부터 읽는다."""
    rows, processed, off, crashed = [], 0, 0, False
    while off < N:
        end = min(off + COMMIT_EVERY, N)
        committed = end                                   # 묶음을 읽자마자 커밋
        for o in range(off, end):
            if not crashed and processed == CRASH_AT:
                crashed = True
                break                                     # 장애: 이 묶음의 나머지는 유실됨
            rows.append((o, ids[o]))
            processed += 1
        off = committed                                   # 재시작은 커밋된 위치부터
    return rows


# A 규칙 실행 및 중복 제거
rows_a = run_rule_a()
rows_raw = len(rows_a)

unique_events_a = set()
dedup_rows = []
for off, eid in rows_a:
    if eid not in unique_events_a:
        unique_events_a.add(eid)
        dedup_rows.append((off, eid))

rows_after_dedup = len(dedup_rows)
duplicates = rows_raw - rows_after_dedup

# B 규칙 실행 및 유실 건수 계산
rows_b = run_rule_b()
unique_events_b = set(eid for off, eid in rows_b)
lost_rule_b = N - len(unique_events_b)

print(f"원천 {N}건, 커밋 주기 {COMMIT_EVERY}, 장애 시점 {CRASH_AT}건 처리 직후")
print(f"A 규칙(처리 후 커밋): 적재 {rows_raw}행, 중복 {duplicates}행, 중복 제거 후 {rows_after_dedup}행 (전달 보장: at-least-once)")
print(f"B 규칙(읽은 후 커밋): 적재 {len(rows_b)}행, 유실 {lost_rule_b}건 (전달 보장: at-most-once)")

key("commit_every", COMMIT_EVERY)
key("crash_at", CRASH_AT)
key("rows_raw", rows_raw)
key("duplicates", duplicates)
key("rows_after_dedup", rows_after_dedup)
key("lost_rule_b", lost_rule_b)
```

```python
문제 4 (5주). 늦게 도착한 이벤트는 집계에 들어갔는가 - 수정본
=====================================================
수정 사항:
1. 결함 1 (L55-57): 워터마크 계산 시점 오류.
   Spark Structured Streaming 및 5주차 규칙에 따르면, 각 배치 파일 처리 시 적용되는 워터마크는

   기존 코드는 현재 배치의 최댓값을 먼저 반영하여 워터마크를 성급하게 전진시킨 뒤 현재 배치를 필터링함으로써
   batch2의 LATE_OK가 부당하게 폐기되었습니다.
   -> 배치 행들을 처리할 때는 직전까지의 watermark를 적용하고, 워터마크 갱신은 배치 처리가 끝난 뒤 수행하도록 수정.
2. 결함 2 (L60): 폐기(drop)된 지연 이벤트의 집계 및 보고 누락.
   운영 보고서는 "14건 모두 정상 집계"라고 하였으나, 실제 폐기된 이벤트를 카운트(dropped)하고
   LATE_OK와 LATE_DROP의 판정 결과를 명시적으로 기록/출력하도록 보완.
"""

import json
from collections import defaultdict
from datetime import datetime, timedelta

from common import PRACTICE, key, parse_id, pick_choice

sid = parse_id()
STREAM_DIR = PRACTICE / "chapter5" / "data" / "input" / "stream"
FMT = "%Y-%m-%d %H:%M:%S"
WINDOW_MIN = pick_choice(sid, "q4-window", [5, 10])
DELAY_MIN = pick_choice(sid, "q4-delay", [2, 3, 4, 5, 6, 8, 12, 29, 30, 35])


def load_batches():
    batches = []
    for path in sorted(STREAM_DIR.glob("*.json")):
        rows = []
        for line in path.read_text(encoding="utf-8").splitlines():
            if line.strip():
                row = json.loads(line)
                row["t"] = datetime.strptime(row["event_time"], FMT)
                rows.append(row)
        batches.append((path.name, rows))
    return batches


def window_of(t):
    start = t.replace(minute=(t.minute // WINDOW_MIN) * WINDOW_MIN, second=0, microsecond=0)
    return start, start + timedelta(minutes=WINDOW_MIN)


batches = load_batches()
delay = timedelta(minutes=DELAY_MIN)
counts = defaultdict(int)
max_seen = None
watermark = None

accepted = 0
dropped = 0
decisions = {}

print(f"창 {WINDOW_MIN}분, 지연 허용 {DELAY_MIN}분")

for name, rows in batches:
    # [결함 1 수정] 직전 배치까지 관측된 최대 시각으로 계산된 watermark를 이번 배치에 적용
    wm_in_effect = watermark
    wm_text = f"{wm_in_effect:%H:%M}" if wm_in_effect else "없음(초기)"
    print(f"\n{name}: 이벤트 {len(rows)}건 처리 (적용 watermark: {wm_text})")

    for r in rows:
        w_start, w_end = window_of(r["t"])
        # 창 끝(w_end)이 적용 워터마크 이하이면 지연 폐기
        is_late = (wm_in_effect is not None) and (w_end <= wm_in_effect)
        if is_late:
            dropped += 1
            decisions[r["event_id"]] = "폐기"
            print(f"  [폐기] {r['event_id']} (발생: {r['t']:%H:%M}, 창끝: {w_end:%H:%M} <= wm: {wm_text})")
            continue

        accepted += 1
        counts[(w_start, r["district"])] += 1
        decisions[r["event_id"]] = "수용"
        print(f"  [수용] {r['event_id']} (발생: {r['t']:%H:%M}, 창: {w_start:%H:%M}~{w_end:%H:%M})")

    # [결함 1 수정] 배치의 처리가 완료된 후 다음 배치를 위한 watermark 갱신
    batch_max = max(r["t"] for r in rows)
    max_seen = batch_max if max_seen is None else max(max_seen, batch_max)
    watermark = max_seen - delay
    print(f"  -> 배치 처리 후 관측 최대 시각: {max_seen:%H:%M}, 다음 배치용 watermark: {watermark:%H:%M}")

late_ok_decision = decisions.get("LATE_OK", "미도착")
late_drop_decision = decisions.get("LATE_DROP", "미도착")

print(f"\n집계 완료: 창-구 {len(counts)}행, 이벤트 {accepted}건 반영, 폐기 {dropped}건")
print(f"판정 결과: LATE_OK={late_ok_decision}, LATE_DROP={late_drop_decision}")

key("window_min", WINDOW_MIN)
key("delay_min", DELAY_MIN)
key("accepted", accepted)
key("dropped", dropped)
key("n_rows", len(counts))
key("late_ok", late_ok_decision)
key("late_drop", late_drop_decision)
```

### 수정한 코드 전문 (`q6_fixed.py`)

```python
"""
문제 6 (7주). 훈련 샘플에 붙인 피처 - 수정본
====================================
수정 사항:
1. 결함 원인 분석:
   - 결함 1 (L46-48, L54): 최신값 조인(latest_value) 결함.
     샘플의 기준 시점(as_of)을 무시하고 현재 시점의 가장 최신 피처를 무조건 결합하여,
     기준 시점 이전에는 아직 생성되지 않은 미래의 데이터가 과거 훈련 샘플에 결합되는
     미래 누출(data leakage / time-travel leak, 7건 발생)을 유발함.
   - 결함 2 (L9, L60): 유효 기간(TTL, 3일) 만료 피처 및 확정 전 피처의 무조건 결합.
     기준 시점 이전에 확정된 피처가 없는 샘플(2건)과 가장 최근 확정치로부터 TTL(3일)을 초과한
     오래된 피처(3건)를 걸러내지 않고 모두 조인시켜, "12행 전부에 피처가 붙었으니 정상"이라고 오판함.
2. 결함 수정:
   - 시점 조인(point-in-time / as-of join) 적용:
     ① 기준 시점(as_of) 이전에 확정(event_timestamp <= as_of)된 후보 중 가장 최근 값 탐색
     ② 후보가 없으면 탈락 (n_dropped_no_value)
     ③ 기준 시점과 피처 확정 시점의 차이가 TTL(3일)을 초과하면 탈락 (n_dropped_ttl)
   - 수정 전 최신값 조인 시 미래 값이 결합되었던 샘플 수(n_leaked_future) 계산 및
     시점 조인 후 남은 행 수(n_joined=7), 결합된 피처 건수 합계(sum_count=45) 출력.
"""

import json
from datetime import datetime, timedelta, timezone

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter7" / "data" / "input" / "ch6_daily"
DAYS = ["2026-07-01", "2026-07-02", "2026-07-03"]
TTL = timedelta(days=3)
CDS = ["11440", "11620", "11680"]


def build_feature_rows():
    rows = []
    for day in DAYS:
        summary = json.loads((INPUT_DIR / day / "daily_summary.json").read_text(encoding="utf-8"))
        known_at = datetime.fromisoformat(day + "T00:00:00+00:00") + timedelta(days=1)
        for r in summary["by_region"]:
            rows.append({"lawd_cd": r["lawd_cd"], "covers": day, "event_timestamp": known_at,
                         "complaint_count": int(r["count"])})
    return rows


SAMPLES = []
for i in range(12):
    cd = pick_choice(sid, f"q6-cd-{i}", CDS)
    day = pick_int(sid, f"q6-day-{i}", 0, 9)
    SAMPLES.append((cd, datetime(2026, 7, 1 + day, 9, 0, tzinfo=timezone.utc)))
SAMPLES.sort(key=lambda s: (s[1], s[0]))


def latest_value(rows, cd):
    cands = [r for r in rows if r["lawd_cd"] == cd]
    return max(cands, key=lambda r: r["event_timestamp"]) if cands else None


def point_in_time(rows, cd, as_of):
    """기준 시점 이전에 알려진 값 중 가장 최근 값. TTL을 넘으면 붙이지 않는다."""
    cands = [r for r in rows if r["lawd_cd"] == cd and r["event_timestamp"] <= as_of]
    if not cands:
        return None, "no_value"
    best = max(cands, key=lambda r: r["event_timestamp"])
    age = as_of - best["event_timestamp"]
    if age > TTL:
        return None, "ttl"
    return best, "ok"


rows = build_feature_rows()

# [결함 1 분석] 수정 전 최신값 조인 시 미래 값이 결합될 뻔한 샘플 수 계산
n_leaked_future = 0
for cd, as_of in SAMPLES:
    lv = latest_value(rows, cd)
    if lv and lv["event_timestamp"] > as_of:
        n_leaked_future += 1

# [결함 1 & 2 수정] 시점 조인 수행
joined = []
n_dropped_no_value = 0
n_dropped_ttl = 0

for cd, as_of in SAMPLES:
    hit, status = point_in_time(rows, cd, as_of)
    if status == "no_value":
        n_dropped_no_value += 1
    elif status == "ttl":
        n_dropped_ttl += 1
    else:
        joined.append({"lawd_cd": cd, "as_of": as_of, "row": hit})

n_samples = len(SAMPLES)
n_joined = len(joined)
sum_count = sum(j["row"]["complaint_count"] for j in joined)

print(f"피처 원천 {len(rows)}행, 훈련 샘플 {n_samples}행")
print(f"[수정 전 결함] 미래 데이터 누출 샘플: {n_leaked_future}행")
print(f"[수정 후 시점 조인] 부착 성공: {n_joined}행, 탈락: {n_dropped_no_value + n_dropped_ttl}행 "
      f"(확정값 없음: {n_dropped_no_value}행, TTL 초과: {n_dropped_ttl}행)")
for j in joined:
    print(f"  {j['lawd_cd']}  기준 {j['as_of']:%m-%d %H:%M}  대상일 {j['row']['covers']}  건수 {j['row']['complaint_count']}")
print(f"조인 결과 {n_joined}행, 붙은 민원 건수 합계 {sum_count}")

key("n_samples", n_samples)
key("n_joined", n_joined)
key("n_dropped_no_value", n_dropped_no_value)
key("n_dropped_ttl", n_dropped_ttl)
key("n_leaked_future", n_leaked_future)
key("sum_count", sum_count)
```
